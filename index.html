⭐ Problem vs. Solution

❌ The Problem

Manual checking of saline bottles requires constant nurse vigilance. Missed empty bottles pose severe health risks due to backflow or air embolisms.

✅ The Solution

Saline Sense automates fluid tracking with high-accuracy ultrasonic distance measurements, broadcasting immediate alerts to nursing stations via Telegram.

🎯 Mission & Objective

In high-volume clinical settings, healthcare staff manage multiple patients simultaneously. Our system automates routine monitoring, eliminating manual error and giving nurses peace of mind with real-time digital status updates.

🔬 How It Works

An HC-SR04 ultrasonic sensor situated above the bottle measures sound impulse reflections from the fluid surface. The onboard ESP32 processes distance trends. When fluid depth passes the critical threshold (>10 cm distance to fluid up to 25 cm max limit), visual warnings trigger on the I2C LCD screen, and an encrypted Telegram HTTP alert is dispatched over Wi-Fi.

⚙️ Operational Workflow
01
Fluid Depletion: Saline drops below safe operating levels.
02
Sensor Detection: Ultrasonic sensor reads distance shift.
03
ESP32 Processing: Microcontroller executes level threshold algorithm.
04
Local Alerting: 16×2 LCD renders immediate status shift.
05
Cloud Notification: Automated message dispatches to Telegram bot.

Saline_Sense.ino 
#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

const char* WIFI_SSID = "SRKMHS";
const char* WIFI_PASS = "Srkmhs1900@#";
const char* BOT_TOKEN = "8603203909:AAF6l2a0C_xP2f5lrZ1KClGio2xQeb1liCE";
const char* CHAT_ID = "8709939894";

const int TRIG = 4;
const int ECHO = 14;
const int SDA_PIN = 2;
const int SCL_PIN = 15;

const float LOW_DISTANCE = 10.0;
const int DEBOUNCE = 3;

int lowCount = 0;
bool alertSent = false;
bool wasLow = false;

LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  Serial.begin(115200);
  Wire.begin(SDA_PIN, SCL_PIN);

  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Saline Sense");
  lcd.setCursor(0, 1);
  lcd.print("Initializing...");

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);

  connectWiFi();
  
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("System Ready");
  delay(1500);
}

void loop() {
  float distance = getDistance();
  
  if (distance > 0) {
    lcd.setCursor(0, 0);
    lcd.print("Dist: ");
    lcd.print(distance, 1);
    lcd.print(" cm    ");

    if (distance >= LOW_DISTANCE) {
      lowCount++;
      lcd.setCursor(0, 1);
      lcd.print("Status: WARNING ");
      
      if (lowCount >= DEBOUNCE && !alertSent) {
        alertSent = true;
        wasLow = true;
        lcd.setCursor(0, 1);
        lcd.print("Sending Alert! ");
        sendTelegram(distance);
      }
    } else {
      lowCount = 0;
      alertSent = false;
      lcd.setCursor(0, 1);
      lcd.print("Status: Normal  ");
    }
  }
  delay(500);
}

float getDistance() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  
  long duration = pulseIn(ECHO, HIGH, 30000);
  if (duration == 0) return -1;
  float cm = duration * 0.0343 / 2;
  return (cm > 400 || cm < 2) ? -1 : cm;
}

void connectWiFi() {
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  int tries = 0;
  while (WiFi.status() != WL_CONNECTED && tries < 20) {
    delay(500);
    tries++;
  }
}

bool sendTelegram(float distance) {
  if (WiFi.status() != WL_CONNECTED) return false;
  HTTPClient http;
  String msg = "SALINE ALERT: Fluid distance at " + String(distance, 1) + " cm!";
  String url = "https://api.telegram.org/bot" + String(BOT_TOKEN) + "/sendMessage?chat_id=" + String(CHAT_ID) + "&text=" + msg;
  http.begin(url);
  int code = http.GET();
  http.end();
  return (code == 200);
}
